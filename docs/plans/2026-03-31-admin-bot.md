# Admin Bot Extended Statistics — Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a Telegram bot that channel owners add as admin. When users click "Расширенная статистика" on the web UI, the backend fetches detailed data (comments, members, admin log, invite links) on-demand via Bot API + Telethon and displays it in 4 collapsible blocks.

**Architecture:** Hybrid Bot API (python-telegram-bot) + Telethon. Bot polling detects when added/removed as admin. On-demand endpoint `/api/channels/{username}/extended` fetches data from both clients, caches 1 hour, saves to DB. Frontend shows button + 4 stat blocks or "add bot" banner.

**Tech Stack:** FastAPI, python-telegram-bot 21.x, Telethon 1.38, SQLAlchemy async, Vue 3 + TypeScript + TailwindCSS

**Spec:** `docs/specs/2026-03-31-admin-bot-design.md`

```python
# backend/app/services/bot_client.py
"""Bot API client for extended channel statistics."""

import logging
from telegram import Bot
from telegram.error import TelegramError
from app.core.config import settings

logger = logging.getLogger(__name__)

_bot: Bot | None = None


def get_bot() -> Bot | None:
    """Get Bot instance. Returns None if BOT_TOKEN not configured."""
    global _bot
    if not settings.BOT_TOKEN:
        return None
    if _bot is None:
        _bot = Bot(token=settings.BOT_TOKEN)
    return _bot


def parse_reactions_detail(reactions: list[dict] | None) -> dict[str, int]:
    """Parse Bot API reaction objects into {emoji: count} dict."""
    if not reactions:
        return {}
    result = {}
    for r in reactions:
        emoji = r.get("emoji", "")
        count = r.get("total_count", 0)
        if emoji and count:
            result[emoji] = count
    return result


async def check_bot_is_admin(chat_id: int | str) -> bool:
    """Check if our bot is an admin in the given chat."""
    bot = get_bot()
    if bot is None:
        return False
    try:
        member = await bot.get_chat_member(chat_id=chat_id, user_id=bot.id)
        return member.status in ("administrator", "creator")
    except TelegramError as e:
        logger.warning("Failed to check bot admin status: %s", e)
        return False


async def fetch_bot_channel_data(chat_id: int | str) -> dict:
    """Fetch extended data via Bot API: member count, messages with replies."""
    bot = get_bot()
    if bot is None:
        return {}

    try:
        member_count = await bot.get_chat_member_count(chat_id=chat_id)
    except TelegramError:
        member_count = 0

    return {
        "member_count": member_count,
    }
```

```python
# backend/app/workers/bot_polling.py
"""Background task: poll for bot admin status changes via my_chat_member updates."""

import asyncio
import logging

from telegram import Bot, Update
from telegram.error import TelegramError
from sqlalchemy import select

from app.core.config import settings
from app.core.database import async_session
from app.models.channel import Channel

logger = logging.getLogger(__name__)


async def handle_my_chat_member(update: Update) -> None:
    """Process my_chat_member update — bot added/removed as admin."""
    chat_member = update.my_chat_member
    if chat_member is None:
        return

    chat = chat_member.chat
    new_status = chat_member.new_chat_member.status
    username = chat.username

    if not username:
        logger.info("Bot status changed in chat without username, skipping")
        return

    is_admin = new_status in ("administrator", "creator")
    logger.info("Bot %s in @%s (status: %s)", "added as admin" if is_admin else "removed", username, new_status)

    async with async_session() as db:
        result = await db.execute(
            select(Channel).where(Channel.username == username.lower())
        )
        channel = result.scalar_one_or_none()
        if channel:
            channel.bot_is_admin = is_admin
            await db.commit()
            logger.info("Updated bot_is_admin=%s for @%s", is_admin, username)
        else:
            logger.info("Channel @%s not in DB yet — will be set on first analyze", username)


async def start_bot_polling() -> None:
    """Start polling for bot updates in background."""
    if not settings.BOT_TOKEN:
        logger.info("BOT_TOKEN not set — bot polling disabled")
        return

    bot = Bot(token=settings.BOT_TOKEN)
    offset = 0
    logger.info("Bot polling started")

    while True:
        try:
            updates = await bot.get_updates(
                offset=offset,
                timeout=30,
                allowed_updates=["my_chat_member"],
            )
            for update in updates:
                offset = update.update_id + 1
                await handle_my_chat_member(update)
        except TelegramError as e:
            logger.error("Bot polling error: %s", e)
            await asyncio.sleep(5)
        except Exception as e:
            logger.error("Bot polling unexpected error: %s", e)
            await asyncio.sleep(10)
```

```python
# --- Extended stats helpers ---

EVENT_TYPE_MAP = {
    "ChannelAdminLogEventActionParticipantJoin": "join",
    "ChannelAdminLogEventActionParticipantLeave": "leave",
    "ChannelAdminLogEventActionEditMessage": "edit_message",
    "ChannelAdminLogEventActionDeleteMessage": "delete_message",
    "ChannelAdminLogEventActionUpdatePinned": "pin",
}


def parse_admin_log_event(event) -> dict:
    """Parse a Telethon admin log event into a flat dict."""
    action_class = event.action.__class__.__name__
    event_type = EVENT_TYPE_MAP.get(action_class, action_class)
    return {
        "event_type": event_type,
        "user_id": event.user_id,
        "event_date": event.date,
        "data": None,
    }


def parse_participant(user, participant) -> dict:
    """Parse a Telethon user + participant into a flat dict."""
    return {
        "telegram_user_id": user.id,
        "username": getattr(user, "username", None),
        "first_name": getattr(user, "first_name", None),
        "joined_at": getattr(participant, "date", None),
    }


async def fetch_admin_log(username: str, limit: int = 100) -> list[dict]:
    """Fetch admin log events (join/leave) for a channel."""
    client = await get_client()
    try:
        from telethon.tl.functions.channels import GetAdminLogRequest
        from telethon.tl.types import ChannelAdminLogEventsFilter

        entity = await client.get_entity(username)
        result = await client(GetAdminLogRequest(
            channel=entity,
            q="",
            min_id=0,
            max_id=0,
            limit=limit,
            events_filter=ChannelAdminLogEventsFilter(join=True, leave=True),
            admins=[],
        ))
        return [parse_admin_log_event(e) for e in result.events]
    except Exception as e:
        logger.error("Failed to fetch admin log for %s: %s", username, e)
        return []


async def fetch_participants(username: str, limit: int = 200) -> list[dict]:
    """Fetch channel participants (up to limit)."""
    client = await get_client()
    try:
        entity = await client.get_entity(username)
        participants = await client.get_participants(entity, limit=limit)
        return [
            parse_participant(p, p.participant)
            for p in participants
            if hasattr(p, "participant")
        ]
    except Exception as e:
        logger.error("Failed to fetch participants for %s: %s", username, e)
        return []


async def fetch_invite_links(username: str) -> list[dict]:
    """Fetch exported invite links for a channel."""
    client = await get_client()
    try:
        from telethon.tl.functions.messages import GetExportedChatInvitesRequest

        entity = await client.get_entity(username)
        me = await client.get_me()
        result = await client(GetExportedChatInvitesRequest(
            peer=entity,
            admin_id=me,
            limit=50,
            revoked=False,
        ))
        return [
            {
                "link": inv.link,
                "creator_id": getattr(inv, "admin_id", None),
                "usage_count": inv.usage or 0,
            }
            for inv in result.invites
        ]
    except Exception as e:
        logger.error("Failed to fetch invite links for %s: %s", username, e)
        return []
```

```python
# backend/app/services/extended_stats.py
"""Orchestrator: combines Bot API + Telethon data for extended stats."""

import asyncio
import logging
from datetime import datetime, timedelta, timezone

from sqlalchemy import delete, func, select
from sqlalchemy.ext.asyncio import AsyncSession

from app.core.config import settings
from app.models.admin_log import ChannelAdminLog
from app.models.channel import Channel
from app.models.invite_link import ChannelInviteLink
from app.models.member import ChannelMember
from app.models.post import Post
from app.services.bot_client import check_bot_is_admin, get_bot

logger = logging.getLogger(__name__)

CACHE_TTL = timedelta(hours=1)


def _utcnow() -> datetime:
    return datetime.now(timezone.utc).replace(tzinfo=None)


def is_cache_fresh(fetched_at: datetime | None) -> bool:
    if fetched_at is None:
        return False
    return (_utcnow() - fetched_at) < CACHE_TTL


async def fetch_extended_stats(channel: Channel, db: AsyncSession) -> dict:
    """Main entry point: fetch extended stats, using cache if fresh."""

    # Check cache
    if is_cache_fresh(channel.extended_fetched_at):
        return await _build_response_from_db(channel, db)

    # Check bot is admin
    bot = get_bot()
    if bot is None or not settings.BOT_TOKEN:
        return {"bot_connected": False, "bot_username": "", "instructions": "BOT_TOKEN not configured"}

    is_admin = await check_bot_is_admin(chat_id=f"@{channel.username}")
    if not is_admin:
        bot_me = await bot.get_me()
        return {
            "bot_connected": False,
            "bot_username": bot_me.username,
            "instructions": f"Добавьте @{bot_me.username} в админы канала для расширенной статистики",
        }

    # Fetch fresh data from Bot API + Telethon
    telethon_available = True
    try:
        await _fetch_and_store_telethon(channel, db)
    except Exception as e:
        logger.error("Telethon fetch failed for %s: %s", channel.username, e)
        telethon_available = False

    # Update cache timestamp
    channel.bot_is_admin = True
    channel.extended_fetched_at = _utcnow()
    await db.commit()

    response = await _build_response_from_db(channel, db)
    response["telethon_available"] = telethon_available
    return response


async def _fetch_and_store_telethon(channel: Channel, db: AsyncSession) -> None:
    """Fetch data from Telethon and save to DB."""
    from app.services.telethon_client import (
        fetch_admin_log,
        fetch_invite_links,
        fetch_participants,
    )

    # Admin log
    events = await fetch_admin_log(channel.username)
    await db.execute(delete(ChannelAdminLog).where(ChannelAdminLog.channel_id == channel.id))
    for ev in events:
        db.add(ChannelAdminLog(channel_id=channel.id, **ev))

    await asyncio.sleep(1)  # rate limit

    # Participants
    members = await fetch_participants(channel.username)
    await db.execute(delete(ChannelMember).where(ChannelMember.channel_id == channel.id))
    for m in members:
        db.add(ChannelMember(channel_id=channel.id, **m))

    await asyncio.sleep(1)  # rate limit

    # Invite links
    links = await fetch_invite_links(channel.username)
    await db.execute(delete(ChannelInviteLink).where(ChannelInviteLink.channel_id == channel.id))
    for lnk in links:
        db.add(ChannelInviteLink(channel_id=channel.id, **lnk))

    await db.flush()


async def _build_response_from_db(channel: Channel, db: AsyncSession) -> dict:
    """Build extended stats response from DB data."""

    # Comments (replies)
    posts = (await db.execute(
        select(Post).where(Post.channel_id == channel.id).order_by(Post.replies_count.desc()).limit(100)
    )).scalars().all()

    total_replies = sum(p.replies_count for p in posts)
    avg_replies = round(total_replies / len(posts), 1) if posts else 0
    top_commented = [
        {"post_id": p.telegram_post_id, "text_preview": (p.text or "")[:80], "replies": p.replies_count}
        for p in posts[:3]
        if p.replies_count > 0
    ]

    # Members
    members = (await db.execute(
        select(ChannelMember).where(ChannelMember.channel_id == channel.id).limit(200)
    )).scalars().all()

    week_ago = _utcnow() - timedelta(days=7)
    joins_7d = sum(1 for m in members if m.joined_at and m.joined_at >= week_ago)

    leaves = (await db.execute(
        select(ChannelAdminLog).where(
            ChannelAdminLog.channel_id == channel.id,
            ChannelAdminLog.event_type == "leave",
            ChannelAdminLog.event_date >= week_ago,
        )
    )).scalars().all()
    leaves_7d = len(leaves)

    # Invite links
    invite_links = (await db.execute(
        select(ChannelInviteLink)
        .where(ChannelInviteLink.channel_id == channel.id)
        .order_by(ChannelInviteLink.usage_count.desc())
    )).scalars().all()

    # Post details
    total_saves = sum(p.saves_count for p in posts)
    avg_saves = round(total_saves / len(posts), 1) if posts else 0

    reactions_agg: dict[str, int] = {}
    for p in posts:
        if p.reactions_detail:
            for emoji, count in p.reactions_detail.items():
                reactions_agg[emoji] = reactions_agg.get(emoji, 0) + count

    return {
        "bot_connected": True,
        "telethon_available": True,
        "fetched_at": channel.extended_fetched_at.isoformat() if channel.extended_fetched_at else None,
        "comments": {
            "total": total_replies,
            "avg_per_post": avg_replies,
            "top_posts": top_commented,
        },
        "members": {
            "total": channel.subscribers_count,
            "sample": [
                {"username": m.username, "first_name": m.first_name, "joined_at": m.joined_at.isoformat() if m.joined_at else None}
                for m in members[:50]
            ],
            "joins_7d": joins_7d,
            "leaves_7d": leaves_7d,
            "net_growth_7d": joins_7d - leaves_7d,
        },
        "traffic_sources": {
            "invite_links": [
                {"link": lnk.link, "usage_count": lnk.usage_count, "creator_id": lnk.creator_id}
                for lnk in invite_links
            ],
        },
        "post_details": {
            "avg_saves": avg_saves,
            "avg_replies": avg_replies,
            "reactions_breakdown": dict(sorted(reactions_agg.items(), key=lambda x: -x[1])[:20]),
        },
    }
```

```python
# ---------------------------------------------------------------------------
# GET /api/channels/{username}/extended
# ---------------------------------------------------------------------------

@router.get("/{username}/extended", response_model=ApiResponse)
async def get_extended_stats(
    username: str,
    db: AsyncSession = Depends(get_db),
) -> ApiResponse:
    """On-demand extended statistics via Bot API + Telethon."""
    username = username.lstrip("@").lower()
    result = await db.execute(select(Channel).where(Channel.username == username))
    channel = result.scalar_one_or_none()
    if channel is None:
        return ApiResponse(success=False, error=f"Channel '{username}' not found")

    data = await fetch_extended_stats(channel, db)
    return ApiResponse(success=True, data=data)
```

```typescript
const extendedData = ref<any>(null)

async function loadExtended() {
  if (!channel.value) return
  extendedLoading.value = true
```
